% n.º 12-13: sections 12 pt, subsections 11 pt. Sizes and title colours follow the TFM Word
% template (2025-12-17): chapters orange (iselorange), sections and subsections bordeaux
% (iselpurple), as set by the isel chapstyle.

  {-3.5ex plus -0.5ex minus -.2ex}%

  {-2.5ex plus -0.5ex minus -.2ex}%

  {\iselthesis@sectionfont\bfseries\fontsize{11}{13.5}\selectfont\color{\iselthesis@subsectioncolor}}}
\renewcommand\printchaptertitle[1]{%
  \ifNoChapNumber
    \noindent\begin{tabular}{@{}p{\textwidth}@{}}%
      \color{iselorange}\chaptitlefont\MakeUppercase{#1}%
    \end{tabular}%
  \else
    % n.º 10: number, title and rule stay within the 2.5 cm margins.
    \color{iselorange}%
    \noindent\begin{tabular}{@{}p{1.0cm}@{}p{\dimexpr\textwidth-1.0cm\relax}@{}}%
      \chapnumfont\thechapter%
      & \chaptitlefont\MakeUppercase{#1}%
    \end{tabular}%
    \par\vspace{0.05cm}%
    \noindent\rule{\textwidth}{.5pt}%
  \fi
  \NoChapNumberfalse%
}
